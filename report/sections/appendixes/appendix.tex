\chapter*{Appendix A --- Warehouse DDL Excerpts}
\label{appendix:ddl}
\addcontentsline{toc}{chapter}{Appendix A --- Warehouse DDL Excerpts}

This appendix shows the DDL definitions of four key warehouse
tables referenced in Section~\ref{sec:proposed:dataarch}:
\texttt{dim\_route}, \texttt{dim\_date},
\texttt{fact\_boardings}, and \texttt{agg\_route\_hour}. The
full warehouse build is in \texttt{src/warehouse/builder.py}
when Phase~2 completes; the excerpts below are the design-time
specifications.

\section*{A.1 \texttt{dim\_route}}

\begin{lstlisting}[language=SQL, basicstyle=\scriptsize\ttfamily]
CREATE OR REPLACE TABLE dim_route AS
SELECT
    tr.route_no                                     AS route_id,        -- canonical
    tr.route_name,
    tr.company_id,
    tr.direction_count,
    tr.stop_count,
    tr.distance_km,
    tr.scheduled_trips_per_day,
    gr.route_no                                     AS gps_route_no,    -- join key
    gr.route_no IS NOT NULL                         AS has_gps_data,
    NULL                                            AS archetype,       -- backfilled by UMAP job
    NULL                                            AS is_school_route, -- backfilled
    EXISTS (SELECT 1 FROM dim_school s
            WHERE haversine(tr.lat, tr.lng, s.lat, s.lng) < 800)
                                                    AS is_school_proximate,
    EXISTS (SELECT 1 FROM metro_feeders mf
            WHERE mf.route_no = tr.route_no)        AS is_metro_feeder
FROM ticket_routes tr
LEFT JOIN gps_routes gr ON gr.route_no = tr.route_no;
\end{lstlisting}

\section*{A.2 \texttt{dim\_date}}

\begin{lstlisting}[language=SQL, basicstyle=\scriptsize\ttfamily]
CREATE OR REPLACE TABLE dim_date AS
WITH calendar AS (
    SELECT generate_series('2026-01-01'::DATE, '2026-03-31'::DATE,
                           INTERVAL 1 DAY) AS date
)
SELECT
    date,
    YEAR(date)                                AS year,
    MONTH(date)                               AS month,
    WEEK(date)                                AS week,
    DAYOFWEEK(date)                           AS dow,
    DAYOFWEEK(date) IN (1, 7)                 AS is_weekend,
    date BETWEEN '2026-01-28' AND '2026-02-05' AS is_tet,
    -- April 30 is in GPS window not AFC window; flag conditionally
    date IN ('2025-04-30', '2025-05-01')      AS is_april30,
    CASE
        WHEN date BETWEEN '2026-01-28' AND '2026-02-05' THEN 'holiday'
        WHEN DAYOFWEEK(date) IN (1, 7)                  THEN 'weekend'
        ELSE                                                  'weekday'
    END                                       AS day_type
FROM calendar;
\end{lstlisting}

\section*{A.3 \texttt{fact\_boardings}}

\begin{lstlisting}[language=SQL, basicstyle=\scriptsize\ttfamily]
CREATE OR REPLACE VIEW fact_boardings AS
SELECT
    t.ticket_id,
    DATE(t.time_created_ticket)              AS date_key,
    EXTRACT(HOUR FROM t.time_created_ticket) AS hour,
    tt.route_id,
    t.station_id                             AS stop_id,
    t.ticket_type                            AS fare_class_id,
    p.payment_method                         AS payment_channel_id,
    tt.bus_id                                AS vehicle_id,
    p.final_price                            AS amount,
    p.final_price = 0                        AS is_free_ticket
FROM tickets t
LEFT JOIN payments     p  ON p.ticket_id = t.ticket_id
LEFT JOIN ticket_trips tt ON tt.trip_id = t.trip_id;
\end{lstlisting}

\section*{A.4 \texttt{agg\_route\_hour}}

\begin{lstlisting}[language=SQL, basicstyle=\scriptsize\ttfamily]
CREATE OR REPLACE TABLE agg_route_hour AS
SELECT
    f.route_id,
    f.hour,
    d.day_type,
    COUNT(*)                                      AS boardings,
    COUNT(*) FILTER (WHERE f.fare_class_id LIKE 'HS%' OR
                            f.fare_class_id LIKE 'SV%') AS student_boardings,
    COUNT(DISTINCT f.ticket_id)                   AS unique_tickets,
    SUM(f.amount)                                 AS revenue
FROM fact_boardings f
JOIN dim_date d ON d.date = f.date_key
GROUP BY 1, 2, 3;
\end{lstlisting}
